function docOf(entries:{start:{numerator:number;denominator:number};lengths:{numerator:number;denominator:number}[];midis?:number[]}[]){
    const expected:ExpectedEvent[]=[];
    const document={format:'authored',timing:'order-only',source:'',tuningMidi:[...tuning],capo:0,measureCount:1,
        moments:entries.map((entry,i)=>({id:`p${i}`,index:i,measure:1,column:i,beatOffset:entry.start,
            notes:entry.lengths.map((length,string)=>{
                const midi=entry.midis?.[string]??64+(i%5),noteId=`n${i}:${string}`;
                const start=rational(entry.start.numerator,entry.start.denominator),end=plus(start,rational(length.numerator,length.denominator));
                expected.push({id:noteId,momentId:`p${i}`,noteId,midi,string,start,end});
                return{id:noteId,string,midi,fret:midi-tuning[string],pitchClass:midi%12,source:null,duration:length};
            })}))};
    return{document,expected};
}

function crosscheck(label:string,document:any,events:ExpectedEvent[],capture=false){
    const before=JSON.stringify(document);freeze(document);
    const snapshot=observeTabWrittenTime(document,1);
    assert.equal(snapshot.data.noteCoverage,'complete',label);
    assert.equal(snapshot.data.intervals.length,events.length,label);
    assert.equal(snapshot.data.ownedNoteCount,events.length,label);
    let refCount=0;for(const d of snapshot.data.endpointDeltas)refCount+=d.starts.length+d.ends.length;
    assert.equal(refCount,events.length*2,label);
    const gotIntervals=snapshot.data.intervals.map(n=>({noteId:n.noteId,momentId:n.momentId,midi:n.midi,string:n.string,start:key(parse(n.start)),end:key(parse(n.end))}));
    const wantIntervals=events.map(n=>({noteId:n.noteId,momentId:n.momentId,midi:n.midi,string:n.string,start:key(n.start),end:key(n.end)}));
    assert.deepEqual(gotIntervals,wantIntervals,label);
    const queries=finiteQueries(events),checks:any[]=[];
    if(events.length)queries.unshift(plus(events.reduce((a,b)=>compare(a.start,b.start)<0?a:b).start,rational(-1)));
    for(const query of queries){
        const got=getTabWrittenNotesAt(snapshot,exact(query));
        const want=active(events,query);
        const first=events.length?events.reduce((a,b)=>compare(a.start,b.start)<0?a:b).start:null;
        const last=events.length?events.reduce((a,b)=>compare(a.end,b.end)>0?a:b).end:null;
        const outside=!first||!last||compare(query,first)<0||compare(query,last)>=0;
        if(outside){assert.equal(got.status,'outside-range',label+' '+key(query));assert.equal('notes'in got,false);}
        else{
            assert.equal(got.status,'complete',label+' '+key(query));
            if(got.status==='complete')assert.deepEqual(got.notes.map(n=>[n.noteId,n.momentId,n.string,n.midi]),want.map(n=>[n.noteId,n.momentId,n.string,n.midi]),label+' '+key(query));
        }
        report.oracle.finiteQueries++;
        if(capture)checks.push({at:key(query),status:got.status,expectedNoteIds:want.map(n=>n.noteId),actualNoteIds:got.status==='complete'?got.notes.map(n=>n.noteId):undefined});
    }
    verifyJson(snapshot.data);assert.equal(JSON.stringify(document),before,label+' source mutation');
    if(capture)report.manual.push({id:label,checks,intervalCount:events.length,endpointReferences:refCount});
}

for(const positions of [128,1024]){
    const fixture=docOf(Array.from({length:positions},(_,i)=>({start:f(i,1024),lengths:Array.from({length:6},()=>f(1))})));
    freeze(fixture.document);
    const buildTimes:number[]=[],queryTimes:number[]=[],heapDeltas:number[]=[],jsonSizes:number[]=[],endpointRefs:number[]=[];
    for(let run=0;run<7;run++){
        global.gc?.();const beforeHeap=process.memoryUsage().heapUsed,then=performance.now();
        const value=observeTabWrittenTime(fixture.document,1);buildTimes.push(performance.now()-then);
        heapDeltas.push(process.memoryUsage().heapUsed-beforeHeap);jsonSizes.push(Buffer.byteLength(JSON.stringify(value.data)));
        const refs=value.data.endpointDeltas.reduce((n,e)=>n+e.starts.length+e.ends.length,0);endpointRefs.push(refs);
        assert.equal(refs,fixture.expected.length*2);assert.ok(value.data.endpointDeltas.length<=fixture.expected.length*2);
        assert.equal(value.data.noteCoverage,'complete');assert.equal(value.data.ownedNoteCount,positions*6);
        const points=[rational(0),rational(1,2048),rational(1,2),rational(1023,1024),rational(1),rational(3,2),rational(2047,1024)];
        const qt=performance.now();
        for(const q of points){
            const got=getTabWrittenNotesAt(value,exact(q)),want=active(fixture.expected,q);
            const last=fixture.expected.at(-1)!.end;
            if(compare(q,last)>=0){assert.equal(got.status,'outside-range');continue;}
            assert.equal(got.status,'complete');
            if(got.status==='complete')assert.deepEqual(got.notes.map(n=>n.noteId),want.map(n=>n.noteId));
        }
        queryTimes.push((performance.now()-qt)/points.length);
    }
    const median=(a:number[])=>[...a].sort((a,b)=>a-b)[Math.floor(a.length/2)];
    report.benchmarks.push({positions,notes:positions*6,pattern:'one bar; all notes duration1, starts i/1024; dense growing overlap',samples:7,
        buildMs:buildTimes,buildMedianMs:median(buildTimes),queryWithOraclePerPointMs:queryTimes,queryWithOraclePerPointMedianMs:median(queryTimes),
        heapDeltaBytes:heapDeltas,heapDeltaMedianBytes:median(heapDeltas),jsonBytes:jsonSizes,endpointReferences:endpointRefs,

}
